// ==========================================
// Filter 1: Grayscale Image
// ==========================================
void filter1_Grayscale(Image& img) {
    for (int i = 0; i < img.width; ++i) {
        for (int j = 0; j < img.height; ++j) {
            int avg = (img(i, j, 0) + img(i, j, 1) + img(i, j, 2)) / 3;
            img(i, j, 0) = avg;
            img(i, j, 1) = avg;
            img(i, j, 2) = avg;
        }
    }
}

// ==========================================
// Filter 2: Black and White Image
// ==========================================
void filter2_BlackAndWhite(Image& img) {
    for (int i = 0; i < img.width; ++i) {
        for (int j = 0; j < img.height; ++j) {
            int avg = (img(i, j, 0) + img(i, j, 1) + img(i, j, 2)) / 3;
            int bw = (avg >= 128) ? 255 : 0;
            img(i, j, 0) = bw;
            img(i, j, 1) = bw;
            img(i, j, 2) = bw;
        }
    }
}

// ==========================================
// Filter 3: Invert Image
// ==========================================
void filter3_Invert(Image& img) {
    for (int i = 0; i < img.width; ++i) {
        for (int j = 0; j < img.height; ++j) {
            for (int k = 0; k < img.channels; ++k) {
                img(i, j, k) = 255 - img(i, j, k);
            }
        }
    }
}

// ==========================================
// Filter 4: Merge Images
// ==========================================
Image filter4_Merge(const Image& img1, const Image& img2) {
    int minW = min(img1.width, img2.width);
    int minH = min(img1.height, img2.height);

    Image mergedImg(minW, minH);

    for (int i = 0; i < minW; ++i) {
        for (int j = 0; j < minH; ++j) {
            for (int k = 0; k < img1.channels; ++k) {
                mergedImg(i, j, k) = (img1(i, j, k) + img2(i, j, k)) / 2;
            }
        }
    }
    return mergedImg;
}

// ==========================================
// Filter 5: Flip Image
// ==========================================
void filter5_Flip(Image& img, char mode) {
    Image temp = img;
    for (int i = 0; i < img.width; ++i) {
        for (int j = 0; j < img.height; ++j) {
            for (int k = 0; k < img.channels; ++k) {
                if (mode == 'h' || mode == 'H') {
                    img(i, j, k) = temp(img.width - 1 - i, j, k);
                }
                else if (mode == 'v' || mode == 'V') {
                    img(i, j, k) = temp(i, img.height - 1 - j, k);
                }
            }
        }
    }
}

// ==========================================
// Filter 6: Rotate Image
// ==========================================
void filter6_Rotate(Image& image) {
    int choice;
    cout << "Choose angle (90, 180, 270): ";
    cin >> choice;

    int times = choice / 90;
    for (int t = 0; t < times; ++t) {
        Image rotated(image.height, image.width);
        for (int i = 0; i < image.width; ++i) {
            for (int j = 0; j < image.height; ++j) {
                for (int k = 0; k < image.channels; ++k) {
                    rotated.setPixel(j, image.width - 1 - i, k, image.getPixel(i, j, k));
                }
            }
        }
        image = rotated;
    }
}

// ==========================================
// Filter 7: Lighten and Darken Image
// ==========================================
void filter7_LightenDarken(Image& img, char choice) {
    for (int i = 0; i < img.width; ++i) {
        for (int j = 0; j < img.height; ++j) {
            for (int k = 0; k < img.channels; ++k) {
                if (choice == 'l' || choice == 'L') {
                    img(i, j, k) = min(255, (int)(img(i, j, k) * 1.5));
                }
                else if (choice == 'd' || choice == 'D') {
                    img(i, j, k) = (int)(img(i, j, k) * 0.5);
                }
            }
        }
    }
}

// ==========================================
// Filter 8: Resize Image
// ==========================================
Image filter8_Resize(const Image& img, int newW, int newH) {
    Image resized_image(newW, newH);
    for (int i = 0; i < newW; ++i) {
        for (int j = 0; j < newH; ++j) {
            int orig_i = i * img.width / newW;
            int orig_j = j * img.height / newH;
            for (int c = 0; c < img.channels; ++c) {
                resized_image(i, j, c) = img(orig_i, orig_j, c);
            }
        }
    }
    return resized_image;
}

// ==========================================
// Filter 9: Crop Image
// ==========================================
Image filter9_Crop(const Image& img, int x, int y, int w, int h) {
    int cropW = min(w, img.width - x);
    int cropH = min(h, img.height - y);
    Image croppedImg(cropW, cropH);

    for (int i = 0; i < cropW; ++i) {
        for (int j = 0; j < cropH; ++j) {
            for (int k = 0; k < img.channels; ++k) {
                croppedImg(i, j, k) = img(x + i, y + j, k);
            }
        }
    }
    return croppedImg;
}

// ==========================================
// Filter 10: Detect Image Edges
// ==========================================
void filter10_DetectEdges(Image& img) {
    filter2_BlackAndWhite(img);
    Image temp = img;

    for (int i = 0; i < img.width - 1; ++i) {
        for (int j = 0; j < img.height - 1; ++j) {
            int currentPixel = temp(i, j, 0);
            int rightPixel = temp(i + 1, j, 0);
            int bottomPixel = temp(i, j + 1, 0);

            if (abs(currentPixel - rightPixel) > 100 || abs(currentPixel - bottomPixel) > 100) {
                img(i, j, 0) = 0;
                img(i, j, 1) = 0;
                img(i, j, 2) = 0;
            }
            else {
                img(i, j, 0) = 255;
                img(i, j, 1) = 255;
                img(i, j, 2) = 255;
            }
        }
    }
}

// ==========================================
// Filter 11: Add a Frame to the Image
// ==========================================
void filter11_AddFrame(Image& img, int frameSize) {
    for (int i = 0; i < img.width; ++i) {
        for (int j = 0; j < img.height; ++j) {
            if (i < frameSize || i >= img.width - frameSize || j < frameSize || j >= img.height - frameSize) {
                img(i, j, 0) = 0;
                img(i, j, 1) = 0;
                img(i, j, 2) = 255;
            }
        }
    }
}

// ==========================================
// Filter 12: Blur Image
// ==========================================
void filter12_Blur(Image& img) {
    Image temp = img;
    int blurSize = 7;
    int radius = blurSize / 2;

    for (int i = radius; i < img.width - radius; ++i) {
        for (int j = radius; j < img.height - radius; ++j) {
            for (int k = 0; k < img.channels; ++k) {
                int sum = 0;
                for (int dx = -radius; dx <= radius; ++dx) {
                    for (int dy = -radius; dy <= radius; ++dy) {
                        sum += temp(i + dx, j + dy, k);
                    }
                }
                img(i, j, k) = sum / (blurSize * blurSize);
            }
        }
    }
}

// ==========================================
// Filter 13: Sunlight Effect
// ==========================================
void filter13_Sunlight(Image& img) {
    for (int i = 0; i < img.width; ++i) {
        for (int j = 0; j < img.height; ++j) {
            img(i, j, 0) = min(255, (int)(img(i, j, 0) * 1.3));
            img(i, j, 1) = min(255, (int)(img(i, j, 1) * 1.15));
            img(i, j, 2) = (int)(img(i, j, 2) * 0.8);
        }
    }
}

// ==========================================
// Filter 14: Oil Painting Effect
// ==========================================
void filter14_OilPainting(Image& img) {
    Image temp = img;
    int radius = 2;

    for (int i = radius; i < img.width - radius; ++i) {
        for (int j = radius; j < img.height - radius; ++j) {
            int intensityCount[20] = { 0 };
            int sumR[20] = { 0 }, sumG[20] = { 0 }, sumB[20] = { 0 };

            for (int dx = -radius; dx <= radius; ++dx) {
                for (int dy = -radius; dy <= radius; ++dy) {
                    int r = temp(i + dx, j + dy, 0);
                    int g = temp(i + dx, j + dy, 1);
                    int b = temp(i + dx, j + dy, 2);

                    int curIntensity = ((r + g + b) / 3) * 19 / 255;
                    intensityCount[curIntensity]++;
                    sumR[curIntensity] += r;
                    sumG[curIntensity] += g;
                    sumB[curIntensity] += b;
                }
            }

            int maxIndex = 0;
            int maxCount = 0;
            for (int idx = 0; idx < 20; ++idx) {
                if (intensityCount[idx] > maxCount) {
                    maxCount = intensityCount[idx];
                    maxIndex = idx;
                }
            }

            img(i, j, 0) = sumR[maxIndex] / maxCount;
            img(i, j, 1) = sumG[maxIndex] / maxCount;
            img(i, j, 2) = sumB[maxIndex] / maxCount;
        }
    }
}

// ==========================================
// Filter 15: TV / Scanlines Effect
// ==========================================
void filter15_TVEffect(Image& img) {
    for (int j = 0; j < img.height; ++j) {
        if (j % 4 == 0) {
            for (int i = 0; i < img.width; ++i) {
                for (int k = 0; k < img.channels; ++k) {
                    img(i, j, k) = (int)(img(i, j, k) * 0.4);
                }
            }
        }
    }
}

// ==========================================
// Filter 16: Purple / Infrared Effect
// ==========================================
void filter16_PurpleEffect(Image& img) {
    for (int i = 0; i < img.width; ++i) {
        for (int j = 0; j < img.height; ++j) {
            img(i, j, 0) = min(255, (int)(img(i, j, 0) * 1.25));
            img(i, j, 1) = (int)(img(i, j, 1) * 0.6);
            img(i, j, 2) = min(255, (int)(img(i, j, 2) * 1.35));
        }
    }
}
